เมื่อใช้งานสกิลและได้ผลลัพธ์แล้ว ส่งกลับมาให้ Vider ได้ดังนี้:

from vider_knowledge_loop import submit_skill_external

result = submit_skill_external(
    skill_id = "smart_notes",

    description = "จดบันทึกและจัดหมวดหมู่อัตโนมัติ",

    result_value = 0.80,       # ค่าผลลัพธ์ -1.0 ถึง +1.0

)

print(result["message"])
 
 
 
 
📊 เกณฑ์การประเมิน
 
ระบบจะประเมินอัตโนมัติตามค่าผลลัพธ์ที่ส่งกลับมา:
 
ช่วงค่าผลลัพธ์ การดำเนินการ สถานะ 
⭐ +0.5 ถึง +1.0 ✅ บันทึกเข้าฐานความรู้ — Vider เรียนรู้และพัฒนาต่อ เรียนรู้แล้ว 
⚠️ -0.49 ถึง +0.49 บันทึกประวัติไว้ รอการประเมินเพิ่มเติม รอประเมิน 
❌ -1.0 ถึง -0.5 ข้าม — ไม่บันทึก ไม่เรียนรู้ ไม่ผ่าน 
 
💡 แนวทางการให้คะแนน:
 
- +0.8 ถึง +1.0 = ทำงานได้สมบูรณ์ มีประโยชน์มาก
- +0.5 ถึง +0.79 = ทำงานได้ดี มีประโยชน์
- 0 = ยังไม่ทดสอบ/ไม่แน่ใจ
- -0.5 ถึง -1.0 = มีปัญหา/ใช้งานไม่ได้
 
 
 
💡 ตัวอย่างสกิล
 
📝 SmartNotes — จดบันทึกอัจฉริยะ
 
python
  
from smart_notes import SmartNotes
sn = SmartNotes()

# เพิ่มบันทึก
sn.add("พรุ่งนี้ประชุมลูกค้า")
sn.add("จ่ายค่าไฟวันศุกร์")

# ดูทั้งหมด
print(sn.all())

# ดูแยกหมวดหมู่
print(sn.by_category())

# ค้นหา
print(sn.search("ลูกค้า"))

# ลบ
sn.delete(1)
 
 
ค่าผลลัพธ์: +0.80 ✅ ผ่านเกณฑ์เรียนรู้
